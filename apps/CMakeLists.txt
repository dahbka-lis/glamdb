foreach(dependency IN ITEMS cli11/CMakeLists.txt asio/asio/include/asio.hpp json/single_include/nlohmann/json.hpp)
    if(NOT EXISTS "${PROJECT_SOURCE_DIR}/third_party/${dependency}")
        message(FATAL_ERROR "Initialize the pinned ${dependency} submodule")
    endif()
endforeach()
add_subdirectory("${PROJECT_SOURCE_DIR}/third_party/cli11"
                 "${PROJECT_BINARY_DIR}/third_party/cli11" EXCLUDE_FROM_ALL SYSTEM)
add_library(glamdb_asio INTERFACE)
target_include_directories(glamdb_asio SYSTEM INTERFACE "${PROJECT_SOURCE_DIR}/third_party/asio/asio/include")
target_compile_definitions(glamdb_asio INTERFACE ASIO_STANDALONE ASIO_NO_DEPRECATED)
find_package(Threads REQUIRED)
target_link_libraries(glamdb_asio INTERFACE Threads::Threads)
add_subdirectory(transport)
add_subdirectory(cli)
if(GLAMDB_BUILD_STORAGE)
    add_subdirectory(server)
endif()
