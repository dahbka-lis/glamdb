cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GlamDBClient VERSION 0.1.0 LANGUAGES CXX)
    include(GNUInstallDirs)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    get_filename_component(GLAMDB_SOURCE_ROOT "../.." ABSOLUTE)
    add_subdirectory("${GLAMDB_SOURCE_ROOT}/third_party/cli11" "${CMAKE_BINARY_DIR}/cli11" EXCLUDE_FROM_ALL SYSTEM)
    find_package(Threads REQUIRED)
    add_library(glamdb_asio INTERFACE)
    target_include_directories(glamdb_asio SYSTEM INTERFACE "${GLAMDB_SOURCE_ROOT}/third_party/asio/asio/include")
    target_compile_definitions(glamdb_asio INTERFACE ASIO_STANDALONE ASIO_NO_DEPRECATED)
    target_link_libraries(glamdb_asio INTERFACE Threads::Threads)
    add_subdirectory("${GLAMDB_SOURCE_ROOT}/apps/client" "${CMAKE_BINARY_DIR}/client")
else()
    set(GLAMDB_SOURCE_ROOT "${PROJECT_SOURCE_DIR}")
endif()
add_executable(glamdb_cli main.cpp)
set_target_properties(glamdb_cli PROPERTIES OUTPUT_NAME glamdb-cli)
target_include_directories(glamdb_cli SYSTEM PRIVATE "${GLAMDB_SOURCE_ROOT}/third_party/json/single_include")
target_link_libraries(glamdb_cli PRIVATE glamdb_client CLI11::CLI11)
if(TARGET glamdb_build_options)
    target_link_libraries(glamdb_cli PRIVATE glamdb_build_options)
endif()
install(TARGETS glamdb_cli RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
